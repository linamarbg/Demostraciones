Apéndice: código de verificación numérica#

El script lars_check.py verifica numéricamente los resultados del capítulo 3: la umbralización suave en el caso ortogonal, el descenso por coordenadas y el algoritmo LARS con la modificación lasso, comparados con scikit-learn. Para reproducirlo basta con python lars_check.py (requiere numpy y scikit-learn).

Código#

  1"""Verificación numérica: LARS con modificación lasso, descenso por coordenadas y soft-thresholding."""
  2import numpy as np
  3from sklearn.datasets import load_diabetes
  4from sklearn.linear_model import Lasso, lars_path
  5
  6
  7def lars_lasso(X, y, tol=1e-12):
  8    """Camino lasso exacto por LARS. Columnas de X con norma 1 y y centrada.
  9    Devuelve lista de (C, beta, n_drops): C = max|x_j^T r| en cada nudo (lambda = 2C)."""
 10    n, p = X.shape
 11    beta = np.zeros(p)
 12    c = X.T @ y
 13    active = [int(np.argmax(np.abs(c)))]
 14    knots = []
 15    drops = 0
 16    for _ in range(10 * p):
 17        C = np.max(np.abs(c))
 18        knots.append((C, beta.copy()))
 19        if C < 1e-10:
 20            break
 21        XA = X[:, active]
 22        s = np.sign(c[active])
 23        Ginv_s = np.linalg.solve(XA.T @ XA, s)
 24        A = 1.0 / np.sqrt(s @ Ginv_s)
 25        w = A * Ginv_s                      # direccion de los coeficientes activos
 26        u = XA @ w                          # vector equiangular (norma 1)
 27        a = X.T @ u
 28        inactive = [k for k in range(p) if k not in active]
 29        gamma, new = C / A, None            # si no hay inactivas: se llega a OLS
 30        for k in inactive:
 31            for g in ((C - c[k]) / (A - a[k]), (C + c[k]) / (A + a[k])):
 32                if tol < g < gamma:
 33                    gamma, new = g, k
 34        drop = None
 35        for i, j in enumerate(active):      # modificacion lasso: cruce por cero
 36            g = -beta[j] / w[i]
 37            if tol < g < gamma:
 38                gamma, drop, new = g, j, None
 39        beta[active] += gamma * w
 40        c = X.T @ (y - X @ beta)
 41        if drop is not None:
 42            beta[drop] = 0.0
 43            active.remove(drop)
 44            drops += 1
 45        elif new is not None:
 46            active.append(new)
 47        else:
 48            knots.append((np.max(np.abs(c)), beta.copy()))
 49            break
 50    return knots, drops

 53def soft(z, t):
 54    return np.sign(z) * np.maximum(np.abs(z) - t, 0.0)
 55
 56
 57def cd_lasso(X, y, lam, iters=5000):
 58    """Descenso por coordenadas para ||y-Xb||^2 + lam*|b|_1 con sum x_ij^2 = n."""
 59    n, p = X.shape
 60    b = np.zeros(p)
 61    for _ in range(iters):
 62        for j in range(p):
 63            r = y - X @ b + X[:, j] * b[j]
 64            b[j] = soft(X[:, j] @ r / n, lam / (2 * n))
 65    return b
 66
 67
 68# ---------- 1) Orto-normal: soft-thresholding vs ridge ----------
 69rng = np.random.default_rng(0)
 70n, p = 200, 5
 71Q, _ = np.linalg.qr(rng.normal(size=(n, p)))
 72Xo = np.sqrt(n) * Q                     # X^T X = n I
 73yo = Xo @ np.array([3.0, -2.0, 0.5, 0.0, 0.0]) + rng.normal(size=n)
 74yo -= yo.mean()
 75lam = 150.0
 76b_ols = Xo.T @ yo / n
 77b_soft = soft(b_ols, lam / (2 * n))
 78b_sk = Lasso(alpha=lam / (2 * n), fit_intercept=False, tol=1e-12, max_iter=100000).fit(Xo, yo).coef_
 79b_ridge = np.linalg.solve(Xo.T @ Xo + lam * np.eye(p), Xo.T @ yo)
 80print("1) ortonormal  max|soft - sklearn| =", np.abs(b_soft - b_sk).max())
 81print("   ridge = ols*n/(n+lam):", np.allclose(b_ridge, b_ols * n / (n + lam)))
 82
 83# ---------- 2) Descenso por coordenadas (columnas con sum x^2 = n) ----------
 84X = load_diabetes().data
 85X = X - X.mean(0)
 86X = X / np.linalg.norm(X, axis=0)       # norma 1
 87y = load_diabetes().target
 88y = y - y.mean()
 89Xn = X * np.sqrt(len(y))                # sum x_ij^2 = n
 90lam = 2 * 0.15 * np.abs(X.T @ y).max()
 91b_cd = cd_lasso(Xn, y, lam)
 92b_sk = Lasso(alpha=lam / (2 * len(y)), fit_intercept=False, tol=1e-12, max_iter=100000).fit(Xn, y).coef_
 93print("2) coord. descent max|cd - sklearn| =", np.abs(b_cd - b_sk).max())
 94
 95# ---------- 3) LARS-lasso vs sklearn.lars_path ----------
 96knots, drops = lars_lasso(X, y)
 97alphas, _, coefs = lars_path(X, y, method="lasso")
 98mine = np.array([b for _, b in knots]).T
 99Cs = np.array([C for C, _ in knots])
100print("3) LARS-lasso: nudos propios =", mine.shape[1], " sklearn =", coefs.shape[1], " cruces por cero =", drops)
101if mine.shape[1] == coefs.shape[1]:
102    # en sklearn, alpha = C / n  (lambda = 2C en la escala de columnas con norma 1)
103    print("   max|C - n*alpha| =", np.abs(Cs - alphas * len(y)).max())
104    print("   max|beta - beta_sklearn| =", np.abs(mine - coefs).max())
105
106# KKT en cada nudo: |2 x_j^T r| = lam para activas (con signo), <= lam para inactivas
107viol = 0.0
108for C, b in knots:
109    g = 2 * X.T @ (y - X @ b)
110    lam_k = 2 * C
111    act = b != 0
112    viol = max(viol, np.abs(g[act] - lam_k * np.sign(b[act])).max() if act.any() else 0,
113               max(0.0, (np.abs(g[~act]) - lam_k).max()) if (~act).any() else 0)
114print("   violacion maxima de KKT en los nudos =", viol)
